# House Telemosaic settings
# Every eligible nativevillages house can receive a basement with
# a telemosaic pad pointing to a distant destination.

# Probability that an eligible house receives the telemosaic basement (0.0 to 1.0)
housetelemosaic_chance (Chance of a telemosaic basement) float 1.0 0.0 1.0

# Ordered list of possible destinations, comma-separated
# (valid values: cave, dungeon, maze, sky)
housetelemosaic_destinations (Possible destinations) string cave,dungeon,maze,sky

# Destination choice mode: ordered (first available from the
# list) or random (random draw among the available ones)
housetelemosaic_dest_mode (Destination choice mode) enum ordered ordered,random

# Also place a return link on the destination pad towards the house
housetelemosaic_return_enabled (Enable return link) bool true

# Place a mini artificial cave if no natural cave is found within
# the search radius
housetelemosaic_cave_artificial_fallback (Artificial fallback cave) bool true

# Target depth of the cave pad below the surface (nodes)
housetelemosaic_cave_depth (Cave depth) int 30 10 200

# Horizontal search radius for a natural cave (nodes)
housetelemosaic_cave_radius (Cave search radius) int 48 16 256

# Depth of the maze pad below the surface (nodes)
housetelemosaic_maze_depth (Maze depth) int 25 5 100

# Width of the generated maze, in cells (the ground footprint
# covers 2W+1 nodes; the maze is square)
housetelemosaic_maze_width (Maze width) int 21 9 51

# Height of the maze corridors (nodes)
housetelemosaic_maze_height (Maze corridor height) int 5 3 10

# Depth of the telemosaic basement below the house floor (nodes)
housetelemosaic_basement_depth (Basement depth) int 5 3 15

# Search radius for a cloudlands island (nodes, 0 to disable)
housetelemosaic_sky_search_radius (Cloudlands search radius) int 512 0 2048

# Maximum number of telemosaic extenders placed per pad (0 = unlimited)
housetelemosaic_max_extenders (Maximum extenders per pad) int 12 0 64

# Detailed debug logging in the server log
housetelemosaic_debug (Debug) bool false
